## 契约（plan-08 §实现规格-2/3 与 plan-04 §交接上下文 同源）

```ts
// POST /api/v1/admin/init/margin — body snake_case（MarginRangePayload），返回 ApiResponse data.task_id
//   成功：{ success: true, data: { task_id }, message }
//   互斥/校验拒绝：{ success: false, data: null, message }（HTTP 200，AdminApiClient 抛错，前端展示 message）

// GET /api/v1/admin/tasks/{task_id} — TaskData camelCase + result（camelCase，前端直消费无二次转换）
interface TaskData {
  taskId: string
  taskType: 'sync_market_margin'
  status: 'pending' | 'running' | 'completed' | 'failed' | 'cancelled'
  progress: number       // 已处理交易日
  total: number          // 交易日口径
  percent: number
  result?: MarginTaskResult | null
  errorMessage?: string | null
  createdAt: string
  // ...其余 AsyncTask camelCase 字段
}

interface MarginTaskResult {
  successCount: number
  skippedCount: number   // 非交易日守卫跳过
  failedCount: number
  dateResults: MarginDateResult[]
  unprocessedDates: string[]   // 取消/超时/重启恢复遗留；完整处理范围时为空
}

interface MarginDateResult {
  tradeDate: string
  status: 'success' | 'failed'
  reason?: string       // 失败原因（展示截断 100 字符、title 保留全文）
}

// GET /api/v1/admin/tasks?task_types=sync_market_margin&page=1&page_size=20 — fetcher 解包
//   返回 { tasks: TaskData[], total, page }（tasks 携带 nullable result）
```
